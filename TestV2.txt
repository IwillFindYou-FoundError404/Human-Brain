// =====================================================================// 2. MÔ PHỎNG GIẢI PHẪU MẠNG CONNECTOME TRÊN MÁY TÍNH (BIOPHYSICAL SIMULATION)// =====================================================================class LIFNeuronNode {
    constructor(id, neurotransmitterType) {
        this.id = id;
        this.type = neurotransmitterType; // 'GLUTAMATE' (Kích thích) hoặc 'GABA' (Ức chế)
        this.vMembrane = -70.0;            // Điện thế màng nghỉ cơ bản (mV)
        this.vThreshold = -50.0;           // Ngưỡng bùng nổ xung thần kinh
        this.vReset = -75.0;               // Điện thế hồi phục sau phóng điện
        this.tau = neurotransmitterType === 'GLUTAMATE' ? 20.0 : 15.0; // Hằng số rò rỉ điện tích
        this.synapticFatigue = 1.0;        // Cơ chế điều hòa năng lượng của tế bào đệm Glial
    }

    // Phương trình vi phân Euler giải tích tích lũy dòng điện qua màng nơ-ron
    evaluatePhysics(injectedCurrent, dt) {
        // dv/dt = (-(v - vRest) + R*I) / tau
        const leak = -(this.vMembrane - (-70.0));
        this.vMembrane += ((leak + injectedCurrent * this.synapticFatigue) / this.tau) * dt;

        let fired = false;
        if (this.vMembrane >= this.vThreshold) {
            fired = true;
            this.vMembrane = this.vReset; // Khai hỏa xung điện thành công, ngay lập tức reset màng
            this.synapticFatigue = Math.max(0.2, this.synapticFatigue - 0.15); // Tiêu hao năng lượng khớp
        } else {
            this.synapticFatigue = Math.min(1.0, this.synapticFatigue + 0.01 * dt); // Tế bào đệm hồi phục glycogen
        }
        return fired;
    }
}
// Lớp giả lập hệ thống cấu trúc 4 thùy não liên kết đồ thị mật độ caoclass ConnectomeArchitecture {
    constructor() {
        this.neuronCount = 120; // Giả lập quy mô luồng nơ-ron xử lý cục bộ trên mây
        this.neurons = Array.from({ length: this.neuronCount }, (_, i) => 
            new LIFNeuronNode(i, Math.random() > 0.25 ? 'GLUTAMATE' : 'GABA')
        );

        // Tạo ma trận trọng số kết nối dày đặc không rẽ nhánh (Adjacency Connectome Graph)
        this.adjacencyMatrix = Array.from({ length: this.neuronCount }, () =>
            Array.from({ length: this.neuronCount }, () => (Math.random() * 2 - 1) * 0.3)
        );

        // Ma trận liên kết đầu ra từ nơ-ron chiếu thẳng vào chiều không gian nhúng của từ vựng (Neurons to Hidden Vectors)
        this.projectionWeights = Array.from({ length: this.neuronCount }, () =>
            Array.from({ length: EMBED_DIM }, () => (Math.random() * 2 - 1) * 0.15)
        );
    }

    /**
     * VÒNG LẶP LAN TRUYỀN XUNG ĐIỆN LIÊN TỤC TRONG THẾ GIỚI ẢO (CONNECTOME TICK PIPELINE)
     * Trái tim của hệ thống: Toàn bộ thông số đầu vào từ game được chuyển hóa thành cường độ dòng điện (I)
     */
    forwardConnectomeCycle(externalStressors, tokenInputs, dt = 0.1) {
        let currentArray = new Array(this.neuronCount).fill(0.0);

        // 1. Kích thích các nơ-ron sinh tồn bằng sát thương vật lý của game (Somatic Sensory Shocks)
        const shockIntensity = (externalStressors.damageSustained || 0) * 90.0;
        for (let i = 0; i < 30; i++) currentArray[i] += shockIntensity;

        // 2. Kích thích các nơ-ron nhận thức bằng từ khóa chat của người chơi (Temporal Language Shocks)
        tokenInputs.forEach(tIdx => {
            const mappedNeuronHash = (tIdx * 7) % this.neuronCount;
            currentArray[mappedNeuronHash] += 65.0; // Bơm dòng điện kích thích vào node đại diện cụ thể
        });

        // 3. Chạy thuật toán lan truyền dòng điện tích tụ xuyên suốt ma trận đồ thị (Graph Propagation)
        let spikeFlags = new Array(this.neuronCount).fill(false);
        for (let step = 0; step < 3; step++) { // Chạy 3 bước lan truyền xung điện nội bộ như hệ tuần hoàn flywire
            for (let i = 0; i < this.neuronCount; i++) {
                let internalSynapticSum = 0.0;
                for (let j = 0; j < this.neuronCount; j++) {
                    if (spikeFlags[j]) {
                        // Nếu nơ-ron trước phóng điện, truyền điện tích qua synapse dựa trên trọng số liên kết hình học
                        const weight = this.adjacencyMatrix[j][i];
                        internalSynapticSum += this.neurons[j].type === 'GLUTAMATE' ? weight * 15.0 : weight * (-12.0);
                    }
                }
                const totalInputCurrent = currentArray[i] + internalSynapticSum;
                spikeFlags[i] = this.neurons[i].evaluatePhysics(totalInputCurrent, dt);
            }
        }

        // 4. Chiếu tổng hợp điện thế của các nơ-ron khai hỏa thành Vector ẩn 8 chiều (Latent Hidden Thought Vector)
        let latentThoughtVector = new Array(EMBED_DIM).fill(0.0);
        for (let i = 0; i < this.neuronCount; i++) {
            if (spikeFlags[i]) {
                for (let d = 0; d < EMBED_DIM; d++) {
                    latentThoughtVector[d] += this.projectionWeights[i][d] * (this.neurons[i].vMembrane + 70.0);
                }
            }
        }
        return latentThoughtVector;
    }

    /**
     * THUẬT TOÁN TỰ HỒI QUY SO KHỚP KHÔNG GIAN ĐẦU RA (AUTOREGRESSIVE SOFTMAX GENERATION)
     * Đo khoảng cách ngữ nghĩa giữa Vector tư duy hiện tại với ma trận từ điển để bắn ra từng Token.
     */
    generateLanguageFromCortex(latentThoughtVector) {
        let sentenceTokens = [];

        for (let tokenStep = 0; tokenStep < 8; tokenStep++) {
            let logits = new Array(VOCAB_SIZE).fill(0.0);

            // Tính tích vô hướng (Dot Product Matrix Multiplication)
            for (let i = 0; i < VOCAB_SIZE; i++) {
                let dotProduct = 0.0;
                for (let d = 0; d < EMBED_DIM; d++) {
                    dotProduct += latentThoughtVector[d] * SEMANTIC_EMBEDDING_SPACE[i][d];
                }
                // Áp đặt điểm phạt (Repetition Penalty) tránh lặp từ vô nghĩa
                if (sentenceTokens.includes(i)) dotProduct -= 1.5;
                logits[i] = dotProduct;
            }

            // Hàm kích hoạt Softmax chuyển đổi điểm số logits thành phân phối xác suất
            const maxLogit = Math.max(...logits);
            const exponents = logits.map(l => Math.exp((l - maxLogit) / 0.45)); // Temp = 0.45
            const sumExps = exponents.reduce((a, b) => a + b, 0);
            const probabilities = exponents.map(e => e / (sumExps || 1e-9));

            // Vòng quay Stochastic Roulette bốc token tự do
            let rand = Math.random();
            local cumulative = 0.0;
            let chosenIndex = VOCAB_SIZE - 1;

            for (let i = 0; i < probabilities.length; i++) {
                cumulative += probabilities[i];
                if (rand <= cumulative) {
                    chosenIndex = i;
                    break;
                }
            }

            sentenceTokens.push(chosenIndex);

            // Đẩy ngược token vừa bốc để cập nhật lại Vector tư duy (Context Autoregressive Shift)
            for (let d = 0; d < EMBED_DIM; d++) {
                latentThoughtVector[d] = latentThoughtVector[d] * 0.5 + SEMANTIC_EMBEDDING_SPACE[chosenIndex][d] * 0.5;
            }
        }

        // Dịch chuyển chỉ mục token về chuỗi văn bản hoàn chỉnh tiếng Việt hiển thị lên Roblox
        return sentenceTokens.map(idx => BRAIN_VOCABULARY[idx]).join(" ").replace(/_/g, " ") + ".";
    }
}
